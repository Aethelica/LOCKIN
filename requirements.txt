# mediapipe is pinned to the 0.10.x line deliberately.
#
# mediapipe 1.0.x (currently 1.0.1) crashes on macOS arm64 inside
# TensorsToDetectionsCalculator::Open() -- it unconditionally builds a Metal GPU
# helper, which aborts the whole process with
#   "graph_service.h: Check failed: service_ Service is unavailable"
# before a single frame is processed. Setting delegate=CPU does NOT avoid it.
# 0.10.35 is the newest release without this bug, and its wheel is py3-tagged so
# it installs on Python 3.14. Do not "upgrade" this pin without re-running
# tests/test_detector_smoke.py on macOS.
mediapipe==0.10.35
opencv-python==5.0.0.93
numpy==2.5.2
pytest==9.1.1

# --- LLM intervention layer (Phase 3) ---
#
# anthropic 1.x is the current major line (1.0 moved the SDK to httpx2 and
# dropped Python < 3.10). Installs and runs clean on the 3.14 venv here.
# Only intervention/anthropic_provider.py imports this -- see that file.
anthropic==1.3.0

# Reads ANTHROPIC_API_KEY out of a gitignored .env so the key never has to be
# re-exported in every new terminal, which is a demo-day footgun.
python-dotenv==1.2.3
